// ==========================================
// Entities (Tables)
// ==========================================

Table customer {
  id integer [primary key, ref: < customer_event_map.customer_id, ref: < book_circulation_log_customer_map.customer_id]
  first_name varchar
  last_name varchar
  address varchar
  country_id int [ref: > country.id]
  zip_id int [ref: > zip.id]
  town_id int [ref: > town.id]
}

Table zip {
  id integer [primary key]
  zip_code varchar
}

Table town {
  id integer [primary key]
  town_name varchar
}

Table country {
  id integer [primary key]
  country_name varchar
}

Table event {
  id integer [primary key, ref: < customer_event_map.event_id, ref: < event_book_map.event_id]
  event_type_id integer [ref: > event_type.id]
  event_start_at_dt datetime
  name varchar
}

Table book {
  isbn integer [primary key, ref: < event_book_map.book_id, ref: < author_book_map.book_id]
  book_genre_id integer [ref: > book_genre.id]
  publisher_id integer [ref: > publisher.id]
  title varchar
}

Table event_type {
  id integer [primary key]
  name varchar
}

Table author {
  id integer [primary key, ref: < author_book_map.author_id]
  author_name varchar
}

Table publisher {
  id integer [primary key]
  name varchar
}

Table book_genre {
  id integer [primary key]
  name varchar
}

Table book_circulation_log {
  id integer [primary key, ref: < book_circulation_log_customer_map.book_circulation_log_id]
  book_isbn integer [ref: > book.isbn]
  loan_starts_at_date date
  loan_ends_at_date date
  book_returned_at_date date
}

Table book_stock_log {
  id integer [primary key, ref: < book_circulation_log.id]
  book_location_id integer [unique, ref: - book_location.id]
  is_in_stock boolean
}

Table book_location {
  id integer [primary key]
  book_location_floor_id integer [ref: > book_location_floor.id]
  book_location_shelf_id integer [ref: > book_location_shelf.id]
}

Table book_location_floor {
  id integer [primary key]
  number integer
}

Table book_location_shelf {
  id integer [primary key]
  number integer
}

// ==========================================
// Bridge Tables (For n:m Relationships)
// ==========================================

Table customer_event_map {
  customer_id integer
  event_id integer
}

Table event_book_map {
  event_id int
  book_id int
}

Table author_book_map {
  book_id int
  author_id int
}

Table book_circulation_log_customer_map {
  customer_id int
  book_circulation_log_id int
}
